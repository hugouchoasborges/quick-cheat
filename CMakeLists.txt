cmake_minimum_required(VERSION 3.24)
project(QuickCheat VERSION 0.5.0 LANGUAGES C CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Quick Cheat requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  add_link_options($<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()
add_library(qc_core src/model.cpp src/storage.cpp)
add_library(qc_media src/attachments.cpp)
target_include_directories(qc_media PUBLIC src)
target_link_libraries(qc_media PUBLIC windowscodecs ole32)
add_library(qc_markdown third_party/md4c/md4c.c)
target_include_directories(qc_markdown PUBLIC third_party/md4c)
target_compile_definitions(qc_markdown PUBLIC MD4C_USE_UTF16)
target_include_directories(qc_core PUBLIC src)
target_link_libraries(qc_core PUBLIC shell32 ole32 normaliz user32)
add_library(qc_collections src/collections.cpp)
target_include_directories(qc_collections PUBLIC src)
target_link_libraries(qc_collections PUBLIC qc_core qc_media qc_markdown windowsapp crypt32)
add_library(qc_community src/community/catalog.cpp src/community/files.cpp src/community/http.cpp src/community/json.cpp)
target_include_directories(qc_community PUBLIC src)
target_link_libraries(qc_community PUBLIC qc_collections windowsapp winhttp bcrypt)
option(QC_BUILD_APP "Build the desktop app" ON)
if(QC_BUILD_APP)
  add_executable(QuickCheat WIN32 src/main.cpp src/platform.cpp src/popup.cpp src/popup_geometry.cpp src/shortcut_tiles.cpp src/category_groups.cpp src/editor.cpp src/settings.cpp src/collection_ui.cpp src/community_window.cpp src/website_editor.cpp src/context.cpp src/browser_context.cpp src/dock.cpp src/dock_menu.cpp src/dock_settings.cpp src/workflow_editor.cpp src/workflow_popup.cpp src/markdown_view.cpp src/theme.cpp src/app.rc)
  target_link_libraries(QuickCheat PRIVATE qc_core qc_media qc_markdown qc_collections qc_community comctl32 user32 gdi32 advapi32 comdlg32 dwmapi uxtheme windowsapp uiautomationcore oleaut32)
  target_link_options(QuickCheat PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
  target_include_directories(QuickCheat PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_executable(qc_icon_tests tests/icon_tests.cpp)
  target_link_libraries(qc_icon_tests PRIVATE user32 gdi32)
  add_test(NAME icon_resources COMMAND qc_icon_tests $<TARGET_FILE:QuickCheat>)
endif()
enable_testing()
add_executable(qc_tests tests/core_tests.cpp)
target_link_libraries(qc_tests PRIVATE qc_core)
add_test(NAME core COMMAND qc_tests)
add_executable(qc_large_store_tests tests/large_store_tests.cpp)
target_link_libraries(qc_large_store_tests PRIVATE qc_core)
add_test(NAME large_store COMMAND qc_large_store_tests)
add_executable(qc_theme_tests tests/theme_tests.cpp)
target_link_libraries(qc_theme_tests PRIVATE qc_core)
add_test(NAME themes COMMAND qc_theme_tests)
add_executable(qc_integration tests/integration.cpp)
target_link_libraries(qc_integration PRIVATE qc_core user32)
add_executable(qc_preview tests/preview.cpp)
target_link_libraries(qc_preview PRIVATE qc_collections user32 gdi32 psapi)
